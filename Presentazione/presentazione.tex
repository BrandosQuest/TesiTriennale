\begin{frame}{Funzionalità base del sistema}
    \begin{columns}[T]
        \begin{column}[T]{0.5\textwidth}
            \textbf{Polling sugli host}

            \begin{itemize}
                \item Template \textbf{ICMP Ping} in Zabbix\\ \texttt{PROBLEM} $\longrightarrow$ \texttt{RESOLVED}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Limite}
            \begin{itemize}
                \item Mancanza dell'associazione: \\
                      \textit{problema $\longleftrightarrow$ metadati dell'host}
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.3cm}
    \begin{block}{Soluzione: logs}
        \vspace{0.2cm}
        Cambiamento di stato host $\longrightarrow$ Zabbix trigger action $\longrightarrow$ metadati host in file CSV\\
        \vspace{0.5cm}
        Questo abilita:
        \begin{itemize}
            \item \textbf{Report giornalieri}
            \item \textbf{Report mensili}
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}[fragile]{Autenticazione server to server}
    L'autenticazione verso Interacta è garantita con \textbf{JWT Assertion} e firmata con algoritmo \textbf{RS512}
    \begin{itemize}
        \item \textbf{Firma asimmetrica}
    \end{itemize}
    \vspace{0.3cm}
    \begin{exampleblock}{Codice di firma (Python)}
        \begin{verbatim}
 signed_jwt=f"{b64_header}.{b64_payload}.{b64url_encode(signature)}"
        \end{verbatim}
    \end{exampleblock}
    \textit{Nota:} RS512 garantisce integrità, autenticazione e non ripudio. La confidenzialità del payload è demandata al canale di comunicazione \textit{HTTPS (TLS)}
\end{frame}
