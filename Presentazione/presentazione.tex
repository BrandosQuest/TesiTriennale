% --- SLIDE : LOGICHE DI POLLING E REPORTISTICA ---
\begin{frame}{Funzionalità base del sistema}
    %\framesubtitle{Superare il limite delle Dashboard real-time}
    \begin{columns}[T]
        \begin{column}[T]{0.5\textwidth}
            \textbf{Polling sugli host}

            \begin{itemize}
                \item Template \textbf{ICMP Ping} in Zabbix\\ \texttt{PROBLEM} $\longrightarrow$ \texttt{RESOLVED}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Limite}
            \begin{itemize}
                \item Mancanza dell'associazione: \\
                      \textit{problema $\longleftrightarrow$ metadati dell'host}
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.3cm}
    \begin{block}{Soluzione: logs}
        {\footnotesize
            Zabbix rileva i cambi di stato %(\texttt{PROBLEM/RESOLVED})(da tenere?) 
            e i metadati ralativi all'host in questione vengono salvati nella VM Ubuntu come righe di un file CSV.\\Questo abilita la generazione di:
            \begin{itemize}
                \item \textbf{Report giornalieri:} numero di \texttt{PROBLEM} rilevati, quali non si sono risolti automaticamente e gli host che mostrano oscillazioni
                \item \textbf{Report mensili:} numero di host attivi, la classifica degli host più problematici, e una analisi sulla frequenza dei guasti
            \end{itemize}}
    \end{block}
\end{frame}

% --- SLIDE : SFIDA TECNICA - TAMPER DETECTION ---
\begin{frame}{Monitoraggio Avanzato: Edge AI e Tamper Detection}
    \framesubtitle{Intercettare le intelligenze a bordo camera}
    Oltre al guasto di rete, l'azienda necessitava di rilevare gli \textbf{oscuramenti fisici} delle telecamere (vernice, rami, atti vandalici).
    \vspace{0.3cm}
    \begin{itemize}
        \item Le telecamere Hikvision dispongono di logiche Edge AI (Tamper Detection).
        \item \textbf{Gestione Configurazioni:} Script periodico in Polling per validare l'attivazione dei servizi XML a bordo camera.
        \item \textbf{Inversione del Paradigma:} Non più Polling, ma \textbf{Streaming asincrono}.
    \end{itemize}
\end{frame}

% --- SLIDE : FLUSSO ASINCRONO E ZABBIX SENDER ---
\begin{frame}{Architettura del Flusso Event-Driven}
    \framesubtitle{Gestione delle connessioni HTTP persistenti}
    \begin{enumerate}
        \item \textbf{Middleware Python:} Apre una connessione HTTP persistente (\textit{Stream}) verso l'endpoint \texttt{ISAPI} delle telecamere.
        \item \textbf{Parsing in Real-Time:} La telecamera invia frammenti XML solo quando si verifica un'anomalia. Lo script li intercetta e li analizza istantaneamente.
        \item \textbf{Push verso Zabbix (Trapping):} I dati vengono iniettati in Zabbix tramite \texttt{zabbix\_sender}, utility a riga di comando ad altissime prestazioni.
        \item \textbf{Zabbix Trapper Items:} Item specifici nel template ricevono passivamente il dato, generando i trigger di allarme oscuramento.
    \end{enumerate}
\end{frame}

% --- SLIDE : INTEGRAZIONE BUSINESS - INTERACTA ---
\begin{frame}{Automazione ticketing}
    %\framesubtitle{Dal rilevamento tecnico all'azione operativa}
    L'obiettivo è integrare il sistema creato alla piattaforma cloud aziendale per la gestione dei ticket, \textbf{Interacta}
    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.4\textwidth}
            \textbf{Il flusso:}\\
            \begin{center}
                \textbf{evento}\\[-0.1cm]
                $\downarrow$\\[-0.1cm]
                Zabbix media type: script\\[-0.1cm]
                % $\downarrow$\\[-0.1cm]
                % Macro e Argomenti\\[-0.1cm]
                $\downarrow$\\[-0.1cm]
                elaborazione\\[-0.1cm]
                $\downarrow$\\[-0.1cm]
                \textbf{apertura/chiusura ticket}
            \end{center}
            % Evento - media type script - macro e argomenti - elaborazione - invio richiesta di apertura ticket
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{block}{Logica}
                Grazie agli argomenti passati dal media type: script, lo script può distinguere tra un problema e una risoluzione.\\
                Successivamente tramite le API REST di Interacta:\\
                $\rightarrow$ \textbf{PROBLEM:} apre un nuovo ticket.\\
                $\rightarrow$ \textbf{RESOLVED:} chiude il ticket associato.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- SLIDE : SICUREZZA E CRITTOGRAFIA (JWT) ---
\begin{frame}[fragile]{Autenticazione server to server}
    %\framesubtitle{Protezione delle transazioni API verso il Cloud}
    L'autenticazione verso Interacta è garantita con \textbf{JWT Assertion} e firmata con algoritmo \textbf{RS512}
    \begin{itemize}
        \item \textit{Firma asimmetrica:} chiave privata locale, chiave pubblica conservata sul server Interacta

    \end{itemize}
    \begin{exampleblock}{Codice di firma (Python)}
        \begin{verbatim}
signed_jwt = f"{b64_header}.{b64_payload}.{b64url_encode(signature)}"
        \end{verbatim}
    \end{exampleblock}
    \textit{Nota:} RS512 garantisce integrità autenticazione e non ripudio. La confidenzialità del payload è demandata al canale \textit{HTTPS (TLS)}.
\end{frame}
