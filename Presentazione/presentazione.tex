% --- SLIDE : ZABBIX E DATA INGESTION ---
\begin{frame}{Il Core del Sistema: Zabbix e CMDB}
    \framesubtitle{Dall'inventario statico al provisioning dinamico}
    \begin{itemize}
        \item \textbf{La Piattaforma:} Adozione di \textbf{Zabbix} (Open-Source) per il monitoraggio IT tramite processi di \textit{Auto-discovery} della rete.
        \item \textbf{Estrazione della Conoscenza (Data Ingestion):}
              \begin{itemize}
                  \item I metadati fisici (ID Palo, Coordinate GPS) risiedevano su archivi Excel statici.
                  \item Sviluppo di uno script \textbf{Python} (librerie \texttt{pandas} e \texttt{openpyxl}).
                  \item Utilizzo delle \textbf{API JSON-RPC} di Zabbix per associare automaticamente i metadati agli host scoperti sulla rete.
              \end{itemize}
    \end{itemize}
    \begin{block}{Risultato: Integrazione GIS}
        Le coordinate iniettate via API hanno permesso di abilitare la \textbf{Geomap} nativa (su base OpenStreetMap). L'azienda può ora visualizzare istantaneamente lo stato e la posizione esatta di un guasto sul territorio.
    \end{block}
\end{frame}

% --- SLIDE : LOGICHE DI POLLING E REPORTISTICA ---
\begin{frame}{Rilevamento Guasti e Analisi Storica}
    \framesubtitle{Superare il limite delle Dashboard real-time}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Logica di Rilevamento (Polling)}
            \begin{itemize}
                \item Utilizzo del protocollo \textbf{ICMP} (Ping).
                \item \textit{Business Rule:} Tolleranza di 1 ora prima di generare l'evento \texttt{PROBLEM} (evita falsi positivi per riavvii fisiologici).
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Il limite del Real-Time}
            \begin{itemize}
                \item Al rientro del guasto (\texttt{OK}), la dashboard si pulisce, perdendo lo storico per le analisi direzionali.
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.3cm}
    \begin{block}{Soluzione: Data Logging e Reportistica}
        Un \textit{Custom Script} intercetta ogni cambio di stato (\texttt{PROBLEM/OK}) e lo storicizza su un server Ubuntu. Questo abilita la generazione di:
        \begin{itemize}
            \item \textbf{Report Giornalieri:} Allarmi attivi, fluttuazioni di rete, focus su TVCC.
            \item \textbf{Report Mensili:} Uptime percentuale, classificazione host critici, analisi guasti per fascia oraria.
        \end{itemize}
    \end{block}
\end{frame}

% --- SLIDE : SFIDA TECNICA - TAMPER DETECTION ---
\begin{frame}{Monitoraggio Avanzato: Edge AI e Tamper Detection}
    \framesubtitle{Intercettare le intelligenze a bordo camera}
    Oltre al guasto di rete, l'azienda necessitava di rilevare gli \textbf{oscuramenti fisici} delle telecamere (vernice, rami, atti vandalici).
    \vspace{0.3cm}
    \begin{itemize}
        \item Le telecamere Hikvision dispongono di logiche Edge AI (Tamper Detection).
        \item \textbf{Gestione Configurazioni:} Script periodico in Polling per validare l'attivazione dei servizi XML a bordo camera.
        \item \textbf{Inversione del Paradigma:} Non più Polling, ma \textbf{Streaming asincrono}.
    \end{itemize}
\end{frame}

% --- SLIDE : FLUSSO ASINCRONO E ZABBIX SENDER ---
\begin{frame}{Architettura del Flusso Event-Driven}
    \framesubtitle{Gestione delle connessioni HTTP persistenti}
    \begin{enumerate}
        \item \textbf{Middleware Python:} Apre una connessione HTTP persistente (\textit{Stream}) verso l'endpoint \texttt{ISAPI} delle telecamere.
        \item \textbf{Parsing in Real-Time:} La telecamera invia frammenti XML solo quando si verifica un'anomalia. Lo script li intercetta e li analizza istantaneamente.
        \item \textbf{Push verso Zabbix (Trapping):} I dati vengono iniettati in Zabbix tramite \texttt{zabbix\_sender}, utility a riga di comando ad altissime prestazioni.
        \item \textbf{Zabbix Trapper Items:} Item specifici nel template ricevono passivamente il dato, generando i trigger di allarme oscuramento.
    \end{enumerate}
\end{frame}

% --- SLIDE : INTEGRAZIONE BUSINESS - INTERACTA ---
\begin{frame}{Chiusura del Ciclo: Automazione Ticketing}
    \framesubtitle{Dal rilevamento tecnico all'azione operativa}
    L'ultimo step del progetto è l'interfacciamento con \textbf{Interacta}, la piattaforma cloud aziendale per la gestione dei task.
    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.4\textwidth}
            \textbf{Il Flusso (Custom Media Type):}
            Zabbix esegue uno script locale passando le "Macro" dell'evento:
            \begin{itemize}
                \item \texttt{status} (PROBLEM / RESOLVED)
                \item \texttt{IP}, \texttt{ID\_palo}, \texttt{lat}, \texttt{lon}
                \item \texttt{problem\_ID}
            \end{itemize}
        \end{column}
        \begin{column}{0.6\textwidth}
            \begin{block}{Logica di Business}
                Lo stato dell'evento (\texttt{status}) viene valutato dallo script Python per invocare le API REST di Interacta:\\
                $\rightarrow$ \textbf{PROBLEM:} Apre un nuovo Ticket.\\
                $\rightarrow$ \textbf{RESOLVED:} Chiude il Ticket associato.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- SLIDE : SICUREZZA E CRITTOGRAFIA (JWT) ---
\begin{frame}[fragile]{Sicurezza Server-to-Server: JWT e RS512}
    \framesubtitle{Protezione delle transazioni API verso il Cloud}
    L'autenticazione verso Interacta è garantita dallo standard \textbf{JSON Web Token (JWT Assertion)}.
    \begin{itemize}
        \item \textbf{Firma asimmetrica (RS512):} Chiave privata locale (JSON), chiave pubblica sul cloud.
        \item \textbf{Libreria \texttt{cryptography}:} Struttura Header + Payload, firma RSA + Hash SHA512.
    \end{itemize}
    \begin{exampleblock}{Codice di firma (Python)}
        \begin{verbatim}
signature = private_key.sign(
    signing_input, padding.PKCS1v15(), hashes.SHA512()
)
signed_jwt = f"{b64_header}.{b64_payload}.{b64url_encode(signature)}"
\end{verbatim}
    \end{exampleblock}
    \textit{Nota architetturale:} RS512 garantisce l'Autenticazione e l'Integrità. La confidenzialità del payload è demandata al canale \textbf{HTTPS (TLS)}.
\end{frame}
