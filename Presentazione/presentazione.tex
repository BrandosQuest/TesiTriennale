\begin{frame}
    \frametitle{Definizione del Problema e Obiettivi}

    \textbf{L'Audio Source Separation}
    \begin{itemize}
        \item L'obiettivo è isolare specifici flussi sonori (sorgenti) a partire da un segnale misto (\textit{canzone}).
        \item Si inquadra nel \textbf{Cocktail Party Problem}: la sfida di emulare la capacità dell'orecchio umano di discriminare un suono da altri.
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Scopo del Progetto}
    \begin{itemize}
        \item \textbf{Sperimentazione tradizionale}: implementazione della tecnica \textit{Harmonic-Percussive Separation} (\textbf{HPS}).
        \item \textbf{Deep Learning}: analisi e test del modello \textbf{Demucs}.
        \item \textbf{Valutazione}: confronto dei risultati e delle metriche di separazione.
    \end{itemize}

    \vspace{0.2cm}

\begin{frame}
    \frametitle{Analisi in Frequenza: I Limiti}

    \textbf{Discrete Fourier Transform (DFT):}
    \begin{itemize}
        \item Analizza le frequenze presenti.
        \item \textbf{Problema:} perde il riferimento temporale.
    \end{itemize}

    \vspace{0.3cm}

    \begin{block}{Principio di Indeterminazione}
        Non è possibile conoscere con precisione assoluta e simultanea \textit{quando} avviene un evento sonoro e \textit{quale} frequenza lo compone.
    \end{block}

    \vspace{0.3cm}

    \textbf{Soluzione Necessaria:}
    \begin{itemize}
        \item Serve una mappatura congiunta \textbf{Tempo-Frequenza}.
    \end{itemize}

    \vspace{0.5cm}

\end{frame}

\begin{frame}
    \frametitle{Metodi Statistici: Non-Negative Matrix Factorization (NMF)}
    \begin{block}{Principio Matematico}
        Si scompone lo spettrogramma originale $V$ nel prodotto di due matrici non negative: $V \approx W \cdot H$
    \end{block}

    \begin{itemize}
        \item \textbf{Dictionary Matrix ($W$)}: contiene le basi spettrali o "firme" caratteristiche di ogni sorgente sonora.
        \item \textbf{Activation Matrix ($H$)}: rappresenta l'inviluppo temporale, indicando l'istante di attivazione di ogni componente di $W$.
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Limiti dell'algoritmo}
    \begin{itemize}
        \item \textbf{Instabilità}: l'algoritmo è soggetto al problema dei minimi locali.
        \item \textbf{Parametrizzazione}: il numero di sorgenti $K$ viene definito preventivamente.
        \item \textbf{Sorgenti Astratte}: separazione matematica ma non \textit{semantica}.
    \end{itemize}

\begin{frame}{Mascheramento}
    \begin{columns}[T]

        \begin{column}{0.48\textwidth}
            \textbf{Maschere Tempo-Frequenza:}
            \begin{itemize}
                \item \textbf{Binary:} Decisione netta (0 o 1).
                \item \textbf{Soft:} Rapporto probabilistico tra le energie.
            \end{itemize}

            \vspace{0.3cm}

            \textbf{Il ritorno al Tempo:}
            \begin{itemize}
                \item Applicazione alla STFT originale ($\mathcal{X}$).
                \item \textbf{iSTFT} per ottenere l'audio finale.
            \end{itemize}

            \vspace{0.3cm}
        \end{column}

        \begin{column}{0.5\textwidth}

            \begin{block}{Binary Masking (es. Armonica)}
                \footnotesize
                \begin{equation*}
                    M_h(n, k) =
                    \begin{cases}
                        1, & \text{se } \tilde{\mathcal{Y}}^h(n, k) \ge \tilde{\mathcal{Y}}^p(n, k) \\
                        0, & \text{altrimenti}
                    \end{cases}
                \end{equation*}
            \end{block}

            \vspace{0.2cm}

            \begin{block}{Soft Masking (es. Armonica)}
                \footnotesize
                \begin{equation*}
                    M_h(n, k) = \frac{\tilde{\mathcal{Y}}^h(n,k) + \frac{\epsilon}{2}}{\tilde{\mathcal{Y}}^h(n,k) + \tilde{\mathcal{Y}}^p(n,k) + \epsilon}
                \end{equation*}
            \end{block}

        \end{column}

    \end{columns}

\end{frame}

\begin{frame}{CNN}
    %\vspace{0.2cm}

    \textbf{Convolutional Neural Networks}
    \begin{itemize}
        \item \textbf{Layer Convoluzionale}: estrazione di caratteristiche locali,  \textit{feature map}, tramite lo scorrimento di un \textit{kernel}.
        \item \textbf{Layer di Attivazione}: introduzione della non linearità:  \textbf{ReLU} ($f(x) = \max(0, x)$)
        \item \textbf{Layer di Pooling}: riduzione delle dimensioni spaziali (es. \textit{Max Pooling}) e invarianza spaziale.
    \end{itemize}

    \vspace{0.2cm}

    \textbf{Il problema della profondità}

    Reti molto profonde rischiano la perdita di dettagli spaziali. Soluzione: \textbf{residual block}, in cui l’input originale aggira i layer convoluzionali
    e va a sommarsi direttamente al loro output.

    \textbf{Cosa cambia rispetto alle U-Net}
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Encoder:} il \textit{max pooling} viene sostituito da convoluzioni 1D. La funzione ReLu viene sostituita con attivazioni \textbf{GLU (Gated Linear Unit)} che attuano una decisione dinamica.
        \item \textbf{BiLSTM (Bidirectional
                  Long Short-Term Memory)}: Fornisce al modello una "memoria"(layer BiLSTM), analizzando il brano in entrambe le direzioni per catturare dipendenze e contesto globale (ritmo e melodia).
        \item \textbf{Decoder e Output}: Tramite convoluzioni 1D trasposte e \textit{skip connections}, in fase ascendente, la rete si dirama per generare simultaneamente 4 sorgenti: \textit{Vocals, Bass, Drums, Other}.
    \end{itemize}

    %L'approccio alla realizzazione dell'algoritmo HPS è stato prettamente manuale, non appoggiandosi alla libreira \texttt{librosa}, per stimolare la comprensione di ogni passaggio.
    \begin{columns}
        \column{0.5\textwidth}

        %\textbf{Pipeline "Manuale"}
        \begin{itemize}
            \item \textbf{STFT}: Mappatura nel dominio tempo-frequenza con parametri fissi ($n_{fft}=2048$, $hop=512$).
            \item \textbf{Filtering}: Applicazione dei filtri mediani direzionali, finestra $L_h=31$. Un filtro orizzontale isola le linee armoniche continue
                  ($Y_h$), mentre un filtro verticale estrae gli impulsi percussivi ($Y_p$).
            \item \textbf{iSTFT}: Sintesi finale del segnale nel dominio del tempo.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Code Snippet}
        \begin{block}{Estratti da \texttt{HPS.ipynb}}
            \tiny
            \begin{itemize}
                \item[] \texttt{X = librosa.stft(x, n\_fft=2048, hop\_length=512)}
                \item[] \texttt{Y = np.abs(X)**2}
                      \vspace{0.2cm}
                \item[] \texttt{\# Filtraggio ($L_h=31, L_p=31$)}
                \item[] \texttt{Y\_h = median\_filter(Y, size=(1, 31))}
                \item[] \texttt{Y\_p = median\_filter(Y, size=(31, 1))}
                      \vspace{0.2cm}
                \item[] \texttt{\# Ricostruzione}
                \item[] \texttt{x\_h = librosa.istft(X\_h)}
            \end{itemize}
        \end{block}
    \end{columns}

\begin{frame}{Test di Robustezza: Addizione del Rumore}

    \textbf{Stress Test:}
    \begin{itemize}
        \item \textbf{Il Problema:} Le valutazioni su audio da studio non riflettono la realtà (es. registrazioni live sporche).
        \item \textbf{Metodo:} Iniezione di \textbf{Rumore Bianco Gaussiano} con un Rapporto Segnale-Rumore (SNR).
    \end{itemize}

    \vspace{0.4cm}

    \begin{alertblock}{Le nostre ipotesi}
        \small
        \textbf{HPS:} essendo puramente geometrico, i filtri mediani verranno "accecati" dal rumore. Ci aspettiamo un risultato impreciso e "metallico".\\
        \vspace{0.1cm}
        \textbf{Demucs:} la rete, avendo appreso pattern semantici durante il training, dovrebbe agire da \textit{denoiser} e non avere troppi problemi nella separazione.
    \end{alertblock}

\end{frame}

\begin{frame}{Considerazioni}
    Dopo aver realizzato e utilizzato i due modelli siamo giunti alle seguenti conclusioni:

    \begin{columns}[T]

        \begin{column}{0.5\textwidth}
            \begin{alertblock}{HPS}
                \small
                \begin{itemize}
                    \item \textbf{Pro:} leggero, veloce e deterministico (zero training).
                    \item \textbf{Contro:} assenza di divisione semantica.
                    \item \textbf{Sotto Stress:} i filtri geometrici non riescono a reggere il rumore.
                \end{itemize}
            \end{alertblock}

            \begin{exampleblock}{Demucs}
                \small
                \begin{itemize}
                    \item \textbf{Pro:} alta qualità.
                    \item \textbf{Contro:} architettura "pesante" e computazionalmente esosa.
                    \item \textbf{Sotto Stress:} la rete agisce da \textit{denoiser} garantendo una separazione accettabile delle tracce.
                \end{itemize}
            \end{exampleblock}
